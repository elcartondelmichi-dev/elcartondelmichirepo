import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
from torch_geometric.loader import DataLoader
from torch_geometric.data import Data

from deck_graph_builder import DeckGraphBuilder
from edh_gnn_model import DeckBracketGNN  # Tu clase GNN guardada en model_gnn.py

# ==========================================
# 1. GENERADOR DE DATASET DE MAZOS (EN MEMORIA)
# ==========================================
def crear_dataset_de_prueba(builder: DeckGraphBuilder, num_samples_por_bracket=25):
    print("📦 Generando dataset de grafos en RAM...")
    
    # Pools de cartas representativas por nivel de poder
    pool_bracket_1 = ["Llanowar Elves", "Cultivate", "Giant Growth", "Serra Angel", "Cancel", "Shock"]
    pool_bracket_2 = ["Sol Ring", "Counterspell", "Wrath of God", "Chatterfang, Squirrel General", "Doubling Season", "Swords to Plowshares"]
    pool_bracket_3 = ["Isochron Scepter", "Dramatic Reversal", "Time Warp", "Helm of the Host", "Staff of Domination", "Priest of Titania"]
    pool_bracket_4 = ["Armageddon", "Thousand-Year Storm", "Isochron Scepter", "Dramatic Reversal", "Time Warp", "Sol Ring"]

    dataset = []
    pools = [pool_bracket_1, pool_bracket_2, pool_bracket_3, pool_bracket_4]

    for bracket_label, pool in enumerate(pools):
        for _ in range(num_samples_por_bracket):
            # Muestreo aleatorio dentro del pool
            deck_sample = np.random.choice(pool, size=len(pool), replace=True).tolist()
            
            # Generar x [N, 22] y edge_index [2, E]
            x, edge_index = builder.build_graph_from_decklist(deck_sample)
            
            # Etiqueta: 0 -> Bracket 1, 1 -> Bracket 2, 2 -> Bracket 3, 3 -> Bracket 4
            y = torch.tensor([bracket_label], dtype=torch.long)
            
            dataset.append(Data(x=x, edge_index=edge_index, y=y))

    return dataset


# ==========================================
# 2. ENTRENAMIENTO E INFERENCIA POR CONSOLA
# ==========================================
def run_all_in_memory():
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"\n⚡ Dispositivo activo: {device}")

    # A) Instanciar Builder, Modelo y Cargar Datos
    builder = DeckGraphBuilder()
    raw_dataset = crear_dataset_de_prueba(builder, num_samples_por_bracket=30) # 120 mazos
    
    train_loader = DataLoader(raw_dataset, batch_size=8, shuffle=True)

    model = DeckBracketGNN(in_channels=22, hidden_channels=64, num_classes=4, dropout_rate=0.2).to(device)
    optimizer = optim.AdamW(model.parameters(), lr=0.001, weight_decay=1e-4)
    criterion = nn.CrossEntropyLoss()

    epochs = 20
    print("\n🚀 --- INICIANDO ENTRENAMIENTO DE LA GNN ---")
    print(f"{'Epoch':^8} | {'Loss Avg':^12} | {'Accuracy':^10}")
    print("-" * 38)
    
    for epoch in range(1, epochs + 1):
        model.train()
        total_loss = 0.0
        correct = 0
        total_samples = 0

        for batch in train_loader:
            batch = batch.to(device)
            optimizer.zero_grad()

            out = model(batch.x, batch.edge_index, batch.batch)
            loss = criterion(out, batch.y)
            loss.backward()
            optimizer.step()

            total_loss += loss.item() * batch.num_graphs
            preds = out.argmax(dim=1)
            correct += (preds == batch.y).sum().item()
            total_samples += batch.num_graphs

        acc = correct / total_samples
        avg_loss = total_loss / total_samples

        if epoch % 2 == 0 or epoch == 1:
            print(f"{epoch:^8d} | {avg_loss:^12.4f} | {acc * 100:^9.1f}%")

    # ==========================================
    # 3. PROBAR INFERENCIA EN VIVO POR CONSOLA
    # ==========================================
    print("\n🔮 --- DEMO DE INFERENCIA EN TIEMPO REAL ---")
    model.eval()

    mazo_test = [
        "Isochron Scepter", "Dramatic Reversal", "Sol Ring", 
        "Counterspell", "Wrath of God", "Time Warp", 
        "Thousand-Year Storm", "Armageddon", "Chatterfang, Squirrel General", 
        "Helm of the Host", "Doubling Season", "Priest of Titania", "Staff of Domination"
    ]

    x, edge_index = builder.build_graph_from_decklist(mazo_test)
    x = x.to(device)
    edge_index = edge_index.to(device)

    with torch.no_grad():
        logits = model(x, edge_index)
        probs = torch.softmax(logits, dim=1).squeeze().cpu().numpy()
        predicted_bracket = torch.argmax(logits, dim=1).item() + 1

    print(f"🃏 Mazo Evaluado ({len(mazo_test)} cartas)")
    print(f"📊 Logits Brutos: {logits.squeeze().cpu().numpy().round(3)}")
    print("📈 Probabilidades por Bracket:")
    for b_idx, p in enumerate(probs, start=1):
        bar = "█" * int(p * 20)
        print(f"   • Bracket {b_idx}: {p*100:5.1f}% {bar}")
    
    print(f"\n🎯 Clasificación Final del Modelo: BRACKET {predicted_bracket}\n")


if __name__ == "__main__":
    run_all_in_memory()
